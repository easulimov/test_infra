#!/bin/bash
# ==============================================================================
# УСТАНОВКА SONATYPE NEXUS REPOSITORY 3
# ==============================================================================

set -u  # Выход при использовании неопределённых переменных

# ==============================================================================
# ЭТАП -1: ПРЕДВАРИТЕЛЬНАЯ ПРОВЕРКА СИСТЕМНЫХ ТРЕБОВАНИЙ SONATYPE
# ==============================================================================

# Проверка прав root
if [ "$EUID" -ne 0 ]; then
    echo "ERROR: Скрипт должен запускаться от имени root (sudo)"
    exit 1
fi

# ВНИМАНИЕ: Замените /dev/sdb на имя вашего свободного диска
TARGET_DISK="/dev/sdb"

# Проверка RAM (минимум 8 GB, рекомендуется 12+ GB)
TOTAL_RAM_MB=$(free -m | awk '/^Mem:/{print $2}')
TOTAL_RAM_GB=$((TOTAL_RAM_MB / 1024))
if [ "$TOTAL_RAM_GB" -lt 8 ]; then
    echo "ERROR: Минимум 8 GB RAM требуется по требованиям Sonatype. У вас: ${TOTAL_RAM_GB} GB"
    exit 1
elif [ "$TOTAL_RAM_GB" -lt 12 ]; then
    echo "WARNING: Sonatype рекомендует 12+ GB RAM. У вас: ${TOTAL_RAM_GB} GB"
fi

# Проверка CPU (минимум 4 ядра, рекомендуется 8+)
CPU_COUNT=$(nproc)
if [ "$CPU_COUNT" -lt 4 ]; then
    echo "ERROR: Минимум 4 CPU требуется. У вас: ${CPU_COUNT}"
    exit 1
elif [ "$CPU_COUNT" -lt 8 ]; then
    echo "WARNING: Рекомендуется 8+ CPU для оптимальной производительности. У вас: ${CPU_COUNT}"
fi

# Проверка размера диска (рекомендуется 100-500+ GB)
if ! blockdev --getsize64 "$TARGET_DISK" &>/dev/null; then
    echo "ERROR: Диск $TARGET_DISK не найден"
    exit 1
fi
DISK_SIZE_GB=$(($(blockdev --getsize64 "$TARGET_DISK") / 1024 / 1024 / 1024))
if [ "$DISK_SIZE_GB" -lt 100 ]; then
    echo "WARNING: Sonatype рекомендует 100-500+ GB для хранилища артефактов. У вас: ${DISK_SIZE_GB} GB"
fi

# Проверка, что диск не используется
if mount | grep -q "^${TARGET_DISK}"; then
    echo "ERROR: Диск $TARGET_DISK уже смонтирован!"
    exit 1
fi
if pvs "$TARGET_DISK" &>/dev/null; then
    echo "ERROR: Диск $TARGET_DISK уже является PV в LVM!"
    exit 1
fi

# Проверка доступности портов
for port in 5432 8081; do
    if ss -tlnp | grep -q ":${port} "; then
        echo "ERROR: Порт ${port} уже занят!"
        ss -tlnp | grep ":${port} "
        exit 1
    fi
done

# Динамический расчет параметров в зависимости от RAM
JVM_HEAP_MB=$((TOTAL_RAM_MB / 2))
[ "$JVM_HEAP_MB" -gt 8192 ] && JVM_HEAP_MB=8192  # Не более 8 GB heap
[ "$JVM_HEAP_MB" -lt 2703 ] && JVM_HEAP_MB=2703  # Минимум Nexus
PG_SHARED_BUFFERS_MB=$((TOTAL_RAM_MB / 4))
PG_EFFECTIVE_CACHE_MB=$((TOTAL_RAM_MB * 3 / 4))

echo "=== Параметры конфигурации ==="
echo "RAM: ${TOTAL_RAM_GB} GB | CPU: ${CPU_COUNT} | Disk: ${DISK_SIZE_GB} GB"
echo "JVM Heap: ${JVM_HEAP_MB} MB | PG shared_buffers: ${PG_SHARED_BUFFERS_MB} MB"
echo "=============================="


# ==============================================================================
# ЭТАП 0: СОЗДАНИЕ ВЫДЕЛЕННОГО LVM ДЛЯ ДАННЫХ NEXUS
# ==============================================================================

dnf install -y lvm2

pvcreate "$TARGET_DISK"
vgcreate vg_nexus "$TARGET_DISK"
lvcreate -l 100%FREE -n lv_data vg_nexus

# Форматирование в XFS (оптимально для БД и больших файлов артефактов)
mkfs.xfs -f /dev/vg_nexus/lv_data

mkdir -p /opt/nexus-data

# noatime/nodiratime — критично для производительности БД и хранилища
# nofail — безопасная загрузка, если диск временно недоступен
echo '/dev/vg_nexus/lv_data /opt/nexus-data xfs defaults,noatime,nodiratime,nofail 0 0' >> /etc/fstab
mount -a

AVAILABLE_GB=$(df -BG /opt/nexus-data | tail -1 | awk '{print int($4)}')
echo "Смонтировано ${AVAILABLE_GB} GB свободного пространства"


# ==============================================================================
# ЭТАП 0.5: ОПТИМИЗАЦИЯ ЯДРА ОС И СИСТЕМНЫХ ЛИМИТОВ
# Требования Sonatype: swappiness=1, отключение THP, file descriptors ≥ 65536
# ==============================================================================

# Параметры ядра для Nexus и PostgreSQL
cat << 'EOF' > /etc/sysctl.d/99-nexus.conf
# Минимизация swap (требование Sonatype для Java-приложений)
vm.swappiness = 1
# Разрешить overcommit для Java
vm.overcommit_memory = 1
# Лимит файловых дескрипторов
fs.file-max = 65536
# Увеличение лимита mmap (важно для PostgreSQL)
vm.max_map_count = 262144
# Оптимизация записи на диск
vm.dirty_ratio = 15
vm.dirty_background_ratio = 5
# Сетевая оптимизация
net.core.somaxconn = 65535
EOF
sysctl --system

# Отключение Transparent Huge Pages (THP) — требование Sonatype для JVM
cat << 'EOF' > /etc/systemd/system/disable-thp.service
[Unit]
Description=Disable Transparent Huge Pages (THP) for Nexus/PostgreSQL
DefaultDependencies=no
After=sysinit.target local-fs.target
Before=basic.target

[Service]
Type=oneshot
ExecStart=/bin/sh -c 'echo never > /sys/kernel/mm/transparent_hugepage/enabled'
ExecStart=/bin/sh -c 'echo never > /sys/kernel/mm/transparent_hugepage/defrag'

[Install]
WantedBy=basic.target
EOF
systemctl enable --now disable-thp.service

# Лимиты открытых файлов для пользователя nexus и postgres
cat << 'EOF' > /etc/security/limits.d/99-nexus.conf
nexus   soft   nofile   65536
nexus   hard   nofile   65536
nexus   soft   nproc    32768
nexus   hard   nproc    32768
postgres soft   nofile   65536
postgres hard   nofile   65536
EOF


# ==============================================================================
# ВАРИАНТ 1: ЗАПУСК ЧЕРЕZ PODMAN-COMPOSE
# ==============================================================================

dnf install -y epel-release
dnf install -y podman podman-compose

mkdir -p /opt/nexus-data/container/{nexus,postgres,initdb}

# UID 200 — стандартный пользователь в контейнере Sonatype
chown -R 200:200 /opt/nexus-data/container/nexus
# UID 70 — стандартный пользователь в postgres:alpine (НЕ 999!)
chown -R 70:70 /opt/nexus-data/container/postgres

# Инициализация pg_trgm (обязательно для Nexus 3.71+)
cat << 'EOF' > /opt/nexus-data/container/initdb/init.sql
CREATE EXTENSION IF NOT EXISTS pg_trgm;
EOF

# compose.yaml с лимитами ресурсов, healthcheck и SELinux-метками :Z
cat << EOF > /opt/nexus-data/container/compose.yaml
services:
  postgres:
    image: docker.io/library/postgres:15-alpine
    container_name: nexus-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: nexus
      POSTGRES_USER: nexus
      POSTGRES_PASSWORD: SecretContainerPassword123
    volumes:
      - /opt/nexus-data/container/postgres:/var/lib/postgresql/data:Z
      - /opt/nexus-data/container/initdb:/docker-entrypoint-initdb.d:Z
    ports:
      - "5432:5432"
    deploy:
      resources:
        limits:
          memory: 2G
          cpus: '2.0'
    ulimits:
      nofile:
        soft: 65536
        hard: 65536
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U nexus -d nexus"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s

  nexus:
    image: docker.io/sonatype/nexus3:latest
    container_name: nexus-app
    restart: unless-stopped
    environment:
      # Heap = 50% от лимита памяти контейнера (6G), остальное — off-heap/metaspace
      INSTALL4J_ADD_VM_PARAMS: >-
        -Xms2g -Xmx4g
        -Dnexus.datastore.enabled=true
        -Dnexus.datastore.nexus.type=postgresql
        -Dnexus.datastore.nexus.jdbcUrl=jdbc:postgresql://postgres:5432/nexus
        -Dnexus.datastore.nexus.username=nexus
        -Dnexus.datastore.nexus.password=SecretContainerPassword123
    volumes:
      - /opt/nexus-data/container/nexus:/nexus-data:Z
    ports:
      - "8081:8081"
    deploy:
      resources:
        limits:
          memory: 6G
          cpus: '4.0'
    ulimits:
      nofile:
        soft: 65536
        hard: 65536
    healthcheck:
      test: ["CMD-SHELL", "/bin/bash -c '</dev/tcp/localhost/8081'"]
      interval: 30s
      timeout: 10s
      retries: 5
      start_period: 180s
    depends_on:
      postgres:
        condition: service_healthy
EOF

# Systemd-сервис для автозапуска compose-стека при загрузке
cat << 'EOF' > /etc/systemd/system/nexus-compose.service
[Unit]
Description=Nexus Stack (Podman Compose)
After=network-online.target
Wants=network-online.target
RequiresMountsFor=/opt/nexus-data

[Service]
Type=oneshot
RemainAfterExit=yes
WorkingDirectory=/opt/nexus-data/container
ExecStart=/usr/bin/podman-compose up -d
ExecStop=/usr/bin/podman-compose down
TimeoutStartSec=300
TimeoutStopSec=120

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable --now nexus-compose.service

echo "Вариант 1 запущен. Статус: systemctl status nexus-compose.service"


# ==============================================================================
# ВАРИАНТ 2: ПОЛНАЯ НАТИВНАЯ УСТАНОВКА НА ALMALINUX 9
# ==============================================================================

# Остановите Вариант 1, если оба варианта тестируются на одном хосте:
# systemctl stop nexus-compose.service

# Nexus 3.87+ (актуально на 2026) поставляется с bundled Java 21.
# Системная Java НЕ нужна — её установка может вызвать конфликты.
dnf install -y wget policycoreutils-python-utils

# Установка PostgreSQL 15
dnf install -y postgresql-server postgresql-contrib
postgresql-setup --initdb

# Аутентификация по паролю (scram-sha-256)
sed -i 's|host.*all.*all.*127.0.0.1/32.*ident|host    all    all    127.0.0.1/32    scram-sha-256|g' /var/lib/pgsql/data/pg_hba.conf
sed -i 's|host.*all.*all.*::1/128.*ident|host    all    all    ::1/128    scram-sha-256|g' /var/lib/pgsql/data/pg_hba.conf

# ТЮНИНГ PostgreSQL согласно рекомендациям Sonatype
cat << EOF >> /var/lib/pgsql/data/postgresql.conf

# === Оптимизация для Nexus Repository (требования Sonatype) ===
max_connections = 200
shared_buffers = ${PG_SHARED_BUFFERS_MB}MB
effective_cache_size = ${PG_EFFECTIVE_CACHE_MB}MB
work_mem = 8MB
maintenance_work_mem = 512MB
wal_buffers = 16MB
checkpoint_completion_target = 0.9
# Для SSD-дисков:
random_page_cost = 1.1
effective_io_concurrency = 200
EOF

systemctl enable --now postgresql

# Создание БД и пользователя
sudo -u postgres psql -c "CREATE USER nexus WITH PASSWORD 'SecretNativePassword123';"
sudo -u postgres psql -c "CREATE DATABASE nexus OWNER nexus;"
sudo -u postgres psql -d nexus -c "CREATE EXTENSION IF NOT EXISTS pg_trgm;"

# Системный пользователь nexus
useradd -r -m -d /opt/nexus-home -s /bin/bash nexus 2>/dev/null || true

mkdir -p /opt/nexus /opt/nexus-data/native

# Загрузка дистрибутива (содержит bundled Java 21)
wget -q https://download.sonatype.com/nexus/3/latest-unix.tar.gz -O /tmp/nexus3.tar.gz
tar -xzf /tmp/nexus3.tar.gz -C /opt/nexus --strip-components=1
rm -f /tmp/nexus3.tar.gz

# Настройка nexus.vmoptions
sed -i "s/-Xms[0-9]*m/-Xms${JVM_HEAP_MB}m/g" /opt/nexus/bin/nexus.vmoptions
sed -i "s/-Xmx[0-9]*m/-Xmx${JVM_HEAP_MB}m/g" /opt/nexus/bin/nexus.vmoptions
sed -i 's|-Dkaraf.data=.*|-Dkaraf.data=/opt/nexus-data/native|g' /opt/nexus/bin/nexus.vmoptions
sed -i 's|-Djava.io.tmpdir=.*|-Djava.io.tmpdir=/opt/nexus-data/native/tmp|g' /opt/nexus/bin/nexus.vmoptions
grep -q "^-Dkaraf.log=" /opt/nexus/bin/nexus.vmoptions || \
    echo "-Dkaraf.log=/opt/nexus-data/native/log" >> /opt/nexus/bin/nexus.vmoptions
sed -i 's|-Dkaraf.log=.*|-Dkaraf.log=/opt/nexus-data/native/log|g' /opt/nexus/bin/nexus.vmoptions

# Подключение внешней PostgreSQL
cat << 'EOF' >> /opt/nexus/bin/nexus.vmoptions
-Dnexus.datastore.enabled=true
-Dnexus.datastore.nexus.type=postgresql
-Dnexus.datastore.nexus.jdbcUrl=jdbc:postgresql://127.0.0.1:5432/nexus
-Dnexus.datastore.nexus.username=nexus
-Dnexus.datastore.nexus.password=SecretNativePassword123
EOF

echo 'run_as_user="nexus"' > /opt/nexus/bin/nexus.rc

chown -R nexus:nexus /opt/nexus /opt/nexus-data/native

# SELinux контексты
semanage fcontext -a -t bin_t "/opt/nexus/bin(/.*)?" 2>/dev/null || true
semanage fcontext -a -t var_log_t "/opt/nexus-data/native/log(/.*)?" 2>/dev/null || true
restorecon -R /opt/nexus /opt/nexus-data/native 2>/dev/null || true

# Systemd-сервис с лимитами
cat << 'EOF' > /etc/systemd/system/nexus.service
[Unit]
Description=Nexus Repository Manager OSS (Native)
After=network-online.target postgresql.service
Wants=network-online.target
RequiresMountsFor=/opt/nexus-data

[Service]
Type=forking
LimitNOFILE=65536
LimitNPROC=32768
LimitMEMLOCK=infinity
User=nexus
Group=nexus
ExecStart=/opt/nexus/bin/nexus start
ExecStop=/opt/nexus/bin/nexus stop
Restart=on-failure
TimeoutStartSec=300
TimeoutStopSec=120

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable --now nexus

# Firewall
firewall-cmd --zone=public --add-port=8081/tcp --permanent
firewall-cmd --reload


# ==============================================================================
# ЭТАП ФИНАЛЬНОЙ ПРОВЕРКИ
# ==============================================================================

echo ""
echo "=== Ожидание запуска Nexus (может занять 2-5 минут) ==="
echo "Для Варианта 1: journalctl -fu nexus-compose.service"
echo "Для Варианта 2: journalctl -fu nexus.service"
echo ""
echo "После запуска получите пароль администратора:"
echo "  Вариант 1: cat /opt/nexus-data/container/nexus/admin.password"
echo "  Вариант 2: cat /opt/nexus-data/native/admin.password"
echo ""
echo "Web-интерфейс: http://<HOST>:8081/"
